cmake_minimum_required(VERSION 3.13)
project(gx2draw CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(SDL2 REQUIRED)
add_executable(gx2draw main.cpp)
target_compile_definitions(gx2draw PRIVATE WITH_RHI_GX2)
target_include_directories(gx2draw PRIVATE /src/Sources/Shaders/Generated)
target_link_libraries(gx2draw SDL2::SDL2-static)
wut_create_rpx(gx2draw)
wut_create_wuhb(gx2draw NAME "GX2Draw" SHORTNAME "GX2Draw" AUTHOR "test")
